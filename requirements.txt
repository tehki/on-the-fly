# Runtime dependencies. Development and CI tooling belongs in requirements-dev.txt.
#
# Every entry here passed the Article 12 admission review recorded in docs/adr/.
# Versions are pinned so a build is reproducible in the weak sense (same declared inputs);
# they are not hash-pinned yet, which is a known gap tracked in docs/CODING_AGENT_ADOPTION.md.
#
# sounddevice  MIT  - PortAudio binding for microphone capture. See ADR 0003.
#                     Brings cffi (and pycparser). Ships prebuilt PortAudio binaries on
#                     Windows and macOS; on Linux it needs the system libportaudio2.
#                     Imported only in src/on_the_fly/infrastructure/audio/, never in
#                     domain/, and only lazily.
sounddevice==0.5.6

# faster-whisper  MIT  - local speech recognition (Whisper via CTranslate2). See ADR 0005.
#                        Brings 19 transitive packages, all permissively licensed, and
#                        about 265 MB installed. Imported only in
#                        src/on_the_fly/infrastructure/asr/, never in domain/, and lazily.
#                        Model weights are NOT bundled: they are fetched at runtime and
#                        refused unless they match a digest pinned in
#                        src/on_the_fly/infrastructure/asr/models.py.
# numpy           BSD   - arrives with faster-whisper, but this project passes audio arrays
#                        to the model directly, so it is declared rather than relied on as
#                        a transitive accident.
faster-whisper==1.2.1
numpy==2.5.2

# sherpa-onnx  Apache-2.0  - streaming speech recognition (Zipformer transducers via ONNX
#                            Runtime). See ADR 0008. One transitive package
#                            (sherpa-onnx-core), ~49 MB installed - a fraction of
#                            faster-whisper, and it targets mobile, which is where ADR 0002
#                            goes in phase 2. Imported only in
#                            src/on_the_fly/infrastructure/asr/, and lazily.
sherpa-onnx==1.13.7

# ctranslate2    MIT  - the inference engine for translation. See ADR 0009. It was already
#                       present transitively via faster-whisper, which is not the same as
#                       being admitted: a transitive dependency is another package's
#                       choice, and this project now depends on it directly. Declaring it
#                       adds no bytes and removes the risk of faster-whisper dropping it.
#                       Requires only numpy (already declared) and pyyaml. Imported only in
#                       src/on_the_fly/infrastructure/translation/, and lazily.
# sentencepiece  Apache-2.0 - tokenisation for OPUS-MT. The one genuinely new runtime
#                       dependency this milestone adds, and a small one: verified to have
#                       *no* transitive dependencies at all, no install scripts, and it
#                       ships type stubs. Published by Google. Required because CTranslate2
#                       takes tokens rather than text, and the OPUS-MT release supplies its
#                       vocabulary as sentencepiece models. Imported only in
#                       src/on_the_fly/infrastructure/translation/, and lazily.
ctranslate2==4.8.2
sentencepiece==0.2.2

# onnxruntime  MIT  - the second inference engine, for translation on hardware CTranslate2
#                     cannot reach. See ADR 0018. Promoted from transitive to declared for
#                     the third time in this file and for the same reason: it arrives with
#                     sherpa-onnx, which is another package's choice, and this project now
#                     imports it directly. Adds no bytes. The reason it exists here at all
#                     is that it is the only inference runtime in this stack with official
#                     Android and iOS builds (ADR 0017), at the same 1.29.0 already
#                     installed. Imported only in
#                     src/on_the_fly/infrastructure/translation/, and lazily.
onnxruntime==1.29.0

# av  BSD-3-Clause  - resampling captured audio to the rate the models take. See ADR 0013.
#                     Real capture hardware offers 44.1/48 kHz and refuses 16 kHz, so the
#                     microphone adapter negotiates a rate and converts. Promoted from
#                     transitive to declared for the same reason as ctranslate2: it was
#                     already a hard requirement of faster-whisper (av>=11), which is
#                     another package's choice, and this project now depends on it
#                     directly. No transitive dependencies of its own.
#                     Footprint note: the wheel bundles prebuilt FFmpeg libraries, the same
#                     kind of native surface ADR 0003 recorded for PortAudio. Not new
#                     exposure - it is already installed - but now explicit.
#                     Imported only in src/on_the_fly/infrastructure/audio/, and lazily.
av==18.1.0
