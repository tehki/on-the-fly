# Runtime dependencies. Development and CI tooling belongs in requirements-dev.txt.
#
# Every entry here passed the Article 12 admission review recorded in docs/adr/.
# Versions are pinned so a build is reproducible in the weak sense (same declared inputs);
# they are not hash-pinned yet, which is a known gap tracked in docs/CODING_AGENT_ADOPTION.md.
#
# sounddevice  MIT  - PortAudio binding for microphone capture. See ADR 0003.
#                     Brings cffi (and pycparser). Ships prebuilt PortAudio binaries on
#                     Windows and macOS; on Linux it needs the system libportaudio2.
#                     Imported only in src/on_the_fly/infrastructure/audio/, never in
#                     domain/, and only lazily.
sounddevice==0.5.6

# faster-whisper  MIT  - local speech recognition (Whisper via CTranslate2). See ADR 0005.
#                        Brings 19 transitive packages, all permissively licensed, and
#                        about 265 MB installed. Imported only in
#                        src/on_the_fly/infrastructure/asr/, never in domain/, and lazily.
#                        Model weights are NOT bundled: they are fetched at runtime and
#                        refused unless they match a digest pinned in
#                        src/on_the_fly/infrastructure/asr/models.py.
# numpy           BSD   - arrives with faster-whisper, but this project passes audio arrays
#                        to the model directly, so it is declared rather than relied on as
#                        a transitive accident.
faster-whisper==1.2.1
numpy==2.5.2
